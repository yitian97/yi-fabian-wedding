📸 Extract Google Photos URLs

Instructions:

  1. Open your Google Photos album in a new tab: https://photos.app.goo.gl/9DdHysbo3EMchceG6
  2. Scroll down to load all photos in the album (or wait for them to load)
  3. Open the browser console:
    • Chrome/Edge: Press F12, or Right-click → Inspect → Console tab
    • Firefox: Press F12, or Right-click → Inspect Element → Console tab
    • Safari: Enable Developer menu first, then press Cmd+Option+C
  4. Copy and paste the code below into the console, then press Enter
  5. Copy the output array and paste it into the textarea below
(function() { var photos = []; var images = document.querySelectorAll('img[src*="googleusercontent.com"]'); images.forEach(function(img) { var src = img.src; if (src && src.includes('googleusercontent.com')) { // Convert to higher resolution var highRes = src.replace(/=w\d+-h\d+/, '=w1920-h1080') .replace(/=s\d+/, '=s1920') .replace(/=w\d+$/, '=w1920'); if (!photos.includes(highRes)) { photos.push(highRes); } } }); console.log('✅ Found ' + photos.length + ' photos'); console.log('📋 Copy this array:'); console.log(JSON.stringify(photos, null, 2)); console.log('\n💡 Tip: Right-click on the output above and select "Copy object" or copy the JSON string'); return photos; })();
